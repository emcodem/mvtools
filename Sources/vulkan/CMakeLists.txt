# mvtools_vulkan: the Vulkan build of the GPU backend (same C API as mvtools_cuda, Sources/cuda/mvcuda.h), loaded at run
# time by mvtools2 and cudafilters (dyn_lib.h open_gpu_backend). Needs no Vulkan SDK at run time (only the driver's loader,
# vulkan-1.dll / libvulkan.so.1, opened at run time) and at build time only glslangValidator (Vulkan SDK, the
# glslang-tools package or conda-forge glslang); the Vulkan headers are vendored in third_party/vulkan.
#
# Shaders: shaders/*.comp, compiled to SPIR-V and embedded; x.pix.comp gives x_8 (MVC_PIX8) and x_16.
# Standalone build of the library alone: cmake -S Sources/vulkan -B build_vk
cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(mvtools_vulkan LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

find_program(MVTOOLS_GLSLANG NAMES glslangValidator glslang HINTS ENV VULKAN_SDK PATH_SUFFIXES bin Bin)
if(NOT MVTOOLS_GLSLANG)
  message(WARNING "mvtools_vulkan: glslangValidator not found (Vulkan SDK / glslang-tools; or set MVTOOLS_GLSLANG); not built")
  return()
endif()

file(GLOB MVVK_SHADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/*.comp")
file(GLOB MVVK_SHADER_INCLUDES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/*.glsl")
set(MVVK_SPV_DIR "${CMAKE_CURRENT_BINARY_DIR}/spv")
file(MAKE_DIRECTORY "${MVVK_SPV_DIR}")

set(MVVK_SPV_HEADERS "")
set(MVVK_TABLE_INCLUDES "")
set(MVVK_TABLE_ENTRIES "")
foreach(src ${MVVK_SHADERS})
  get_filename_component(fname "${src}" NAME)
  if(fname MATCHES "^(.*)\\.pix\\.comp$")
    set(base "${CMAKE_MATCH_1}")
    set(variants "8;16")
  else()
    string(REGEX REPLACE "\\.comp$" "" base "${fname}")
    set(variants "-")
  endif()
  foreach(v ${variants})
    if(v STREQUAL "-")
      set(name "${base}")
      set(defs "")
    else()
      set(name "${base}_${v}")
      if(v STREQUAL "8")
        set(defs "-DMVC_PIX8")
      else()
        set(defs "")
      endif()
    endif()
    set(out "${MVVK_SPV_DIR}/${name}.h")
    add_custom_command(OUTPUT "${out}"
      COMMAND "${MVTOOLS_GLSLANG}" -V --target-env vulkan1.2 ${defs} "-I${CMAKE_CURRENT_SOURCE_DIR}/shaders"
              --vn "mvvk_spv_${name}" -o "${out}" "${src}"
      DEPENDS "${src}" ${MVVK_SHADER_INCLUDES}
      COMMENT "SPIR-V ${name}"
      VERBATIM)
    list(APPEND MVVK_SPV_HEADERS "${out}")
    string(APPEND MVVK_TABLE_INCLUDES "#include \"${name}.h\"\n")
    string(APPEND MVVK_TABLE_ENTRIES "  { \"${name}\", mvvk_spv_${name}, sizeof(mvvk_spv_${name}) / 4 },\n")
  endforeach()
endforeach()

set(MVVK_TABLE_CONTENT "// generated by Sources/vulkan/CMakeLists.txt: the embedded SPIR-V shaders\n#include <cstddef>\n#include <cstdint>\n${MVVK_TABLE_INCLUDES}\nstruct MvvkSpv\n{\n  const char *name;\n  const uint32_t *code;\n  size_t words;\n};\nextern const MvvkSpv mvvk_spv_table[] = {\n${MVVK_TABLE_ENTRIES}  { nullptr, nullptr, 0 }\n};\nextern const size_t mvvk_spv_count = sizeof(mvvk_spv_table) / sizeof(mvvk_spv_table[0]) - 1;\n")
file(WRITE "${MVVK_SPV_DIR}/spv_table.cpp.new" "${MVVK_TABLE_CONTENT}")
configure_file("${MVVK_SPV_DIR}/spv_table.cpp.new" "${MVVK_SPV_DIR}/spv_table.cpp" COPYONLY)

file(GLOB MVVK_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.h")
add_library(mvtools_vulkan SHARED ${MVVK_SOURCES} "${MVVK_SPV_DIR}/spv_table.cpp" ${MVVK_SPV_HEADERS})
target_include_directories(mvtools_vulkan PRIVATE "${MVVK_SPV_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(mvtools_vulkan PRIVATE MVCUDA_EXPORTS)
set_target_properties(mvtools_vulkan PROPERTIES
  CXX_STANDARD 17
  CXX_STANDARD_REQUIRED ON
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
  POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set_target_properties(mvtools_vulkan PROPERTIES PREFIX "" MSVC_RUNTIME_LIBRARY "MultiThreaded")
  target_compile_definitions(mvtools_vulkan PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  target_compile_options(mvtools_vulkan PRIVATE /O2 /W3 /wd4100 /wd4244 /wd4267)
elseif(WIN32)
  set_target_properties(mvtools_vulkan PROPERTIES PREFIX "")
  target_link_options(mvtools_vulkan PRIVATE -static-libgcc -static-libstdc++)
else()
  target_compile_options(mvtools_vulkan PRIVATE -O2)
  target_link_libraries(mvtools_vulkan PRIVATE ${CMAKE_DL_LIBS} pthread)
endif()

include(GNUInstallDirs)
install(TARGETS mvtools_vulkan
  LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/avisynth"
  RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
